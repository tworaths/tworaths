from typing import Union, List, Literal  
from enum import IntEnum


class subentity_typing(IntEnum):
    NULL = 0  # i.e. unassigned
    SUBGROUP = 1
    SUBREGION = 2
    CAP = 3  # Capacity-Adequate-Portfolio
    
    @classmethod
    def get_names(cls) -> List[str]:
        return [_ev.name for _ev in cls._value2member_map_.values()]
    
    @classmethod
    def has_name(cls, name) -> Union[Literal[True], Literal[False]]:
        # return name in cls._value2member_map_
        return name in cls.get_names()
