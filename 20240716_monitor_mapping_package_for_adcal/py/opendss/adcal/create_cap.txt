    class analysis_types(IntEnum):

        EUE = 2   # Expected Unserved Energy (in MWh terms) per 1 million MWhs (i.e. 1 TWh) of Annual Delivered Energy. 

    class min_gen_modes(IntEnum):  
        Lenient = 1
        Strict = 2
    
    class maintenance_modes(IntEnum):  
        _None = 0  # No maintenance
        """ Automated maintenance, according to a margin specified in absolute terms. 
        - Objective is to schedule maintenance such that max. capacity margin is achieved each week, in absolute MW terms. """
        AutoAbs = 1  
        """ Automated maintenance, according to a margin specified in percentage terms. 
        - Objective is to schedule maintenance such that max. capacity margin is achieved each week, as a percentage of weekly peak. """
        AutoPC = 2  # Automated maintenance, according to a margin specified in percentage terms
        Explicit = 3  # Explicit maintenance enacted across all units
        MaintenanceRankingByInputOrder = 4
        MaintenanceRankingByCapacityOrder = 5
        MaintenanceRankingByCapacitySOOrder = 6

        Build = 3

def proc_demand_part_one(
        mingen_prop_as_percent: Number=0, 
        mingen_floor_in_mw: Number=0,
        load_incr_val: Number=0, 
        annual_peak_in_MW: Number=0, 
        df_demand_profiles: pd.DataFrame=None,
        b_load_incr_type_is_proportional: Optional[Union[Literal[True], Literal[False]]] = False
    ) -> None:  # -> pd.DataFrame (of MinGens)
    """  Adjust the demand for LoadInc and/or LoadModWB as required. 

    Keyword Args:
        mingen_prop_as_percent(Number):
            - Proportion of original load which sets a minimum below which load must not fall after LoadModWB impact (wind)
        mingen_floor_in_mw(Number):
            - Floor value for Min Generation. A limit, below which, `mingen_prop_as_percent` cannot go.
        annual_peak_in_MW(Number):
        df_demand_profiles(pd.DataFrame):
        b_load_incr_type_is_proportional(Optional[Union[Literal[True], Literal[False]]]):
    """

    """ Create a Min Gen profile/series that is 'like', i.e. same structure, as df_demand_profiles """

    # Choose the appropriate value for `LoadIncrNow`
    _load_incr_frac = None

    """ See Ln 343 of `AdcalV4.10.AdCal2_AdCalDemandFns`, and call to `ProcLoadInc` function. """
    # ProcLoadInc ... to increment loads ... see Ln 410 of `AdcalV4.10.AdCal2_AdCalDemandFns`
    
    ## ProcLoadInc (START) ##
    # annual_energy_in_GWh = np.sum(df_demand_profiles.loc[:, :].values) / 2_000
    logs.debug("\'df_demand_profiles.columns\' (len=%i):\t%s" % 
               (len(df_demand_profiles.columns.values), df_demand_profiles.columns.values))

    # df_mingen_profiles = pd.DataFrame.from_dict({_k: np.zeros(len(df_demand_profiles.index.values)) 
    #                                              for _k in df_demand_profiles.columns.values})
    df_mingen_profiles = df_demand_profiles.copy()
    logs.debug("\'df_mingen_profiles\' [shape=(%i, %i)]\n" % df_mingen_profiles.shape)
    df_mingen_profiles.loc[:, :] *= mingen_prop_as_percent  # Ln 497. For Constant `LoadInc`, MinGen is computed before the increment is applied
    
    if b_load_incr_type_is_proportional:  # Increments in Load are proportional
        _load_incr_frac = load_incr_val / annual_peak_in_MW
        # Apply half-hourly increment in load is: `existing MW` * _load_incr_frac
        df_demand_profiles.loc[:, :] *= (1 + _load_incr_frac)
    else:  # Constant increments in Load are applied (in MW terms)
        df_demand_profiles.loc[:, :] += load_incr_val

    # Need to check that Max Demand is not exceeded (hard set to 100_000 MW in AdCal main)
    if df_demand_profiles.loc[:, :].max(axis=None) > MAX_DEMAND_BY_HALF_HOUR:
        raise ValueError(("Max Demand (half-hourly), {0} MW, cannot exceed the upper limit: {1} MW.\n"
                          ).format(df_demand_profiles.loc[:, :].max(), MAX_DEMAND_BY_HALF_HOUR))
    
    if df_mingen_profiles.loc[:, :].min(axis=None) < mingen_floor_in_mw:  # Assert Min Gen Floor
        for _doy_col in df_mingen_profiles.columns:
            df_mingen_profiles.loc[:, df_mingen_profiles[_doy_col] < mingen_floor_in_mw] = mingen_floor_in_mw

    if not b_load_incr_type_is_proportional: 
        df_mingen_profiles.loc[:, :] += load_incr_val

    """ Update the annual energy and peak MW totals. """
    annual_energy_in_GWh = compute_energy_total(df_demand_profiles, 0.5, 'GWh', None)
    annual_peak_in_MW = get_peak_demand(df_demand_profiles, None)
    ## ProcLoadInc (END) ##

    """ 'Read LoadModWB option. Ln 347, of `AdcalV4.10.AdCal2_AdCalDemandFns` ... LoadModification 
    See \S2.3.4 of AdCal documentation. 

    ... Not urgent just now, but an interesting approach for the modelling of wind. """

    return annual_energy_in_GWh, annual_peak_in_MW, df_mingen_profiles

        logs.info("Week 2, Day 5:\t%s\n" % df_demand_profiles.loc[:, 'wk02d05'])

        # logs.critical("Annual Peak Demand [1] = %.3f MW\n" % np.max(df_demand_profiles.loc[:, :].values))
        logs.critical("Annual Peak Demand [2] = %.3f MW\n" % annual_peak_in_MW)
        
        # logs.debug("Peak Demand for Wk02 [3] = %.3f MW\n" % 
        #            np.max(df_demand_profiles.loc[:, ['wk02d01', 'wk02d02', 'wk02d03', 'wk02d04', 'wk02d05', 'wk02d06', 'wk02d07']].values))
        # logs.debug("Peak Demand for Wk02 [4] = %.3f MW\n" % get_peak_demand(df_demand_profiles, (2, None)))
        
        # logs.debug("Peak Demand for Wk02d05 [5] = %.3f MW\n" % get_peak_demand(df_demand_profiles, (2, 5)))
        # logs.debug("Peak Demand for \*d05 [6] = %.3f MW\n" % get_peak_demand(df_demand_profiles, (None, 5)))
        
        sum_of_all_daily_peaks = np.sum([np.max(df_demand_profiles.loc[:, _doy_col].values) for _doy_col in df_demand_profiles.columns])
        cnt_days_of_year = len(df_demand_profiles.columns)
        seasonal_load_factor = (sum_of_all_daily_peaks / cnt_days_of_year) / annual_peak_in_MW
        logs.critical("Seasonal Load Factor = %.3f \n" % seasonal_load_factor)
        
        total_load_factor = (1_000 * annual_energy_in_GWh) / (annual_peak_in_MW * (24 * cnt_days_of_year))
        logs.critical("Total Load Factor = %.3f \n" % total_load_factor)
        
        daily_load_factor = total_load_factor / seasonal_load_factor
        logs.critical("Daily Load Factor = %.3f \n" % daily_load_factor)

    return annual_energy_in_GWh, annual_peak_in_MW, seasonal_load_factor, total_load_factor, daily_load_factor


def compute_energy_total(df_demand_profiles: pd.DataFrame=None,
                         time_interval_in_hours: Number=0.5,  
                         units: str='GWh', 
                         t_week_and_day_pair: Optional[Tuple[Union[int, None], Union[int, None]]]=None) -> Number:
    """ Default half-hourly, time interval. """
    return (time_interval_in_hours * 
            np.sum(_get_df_profiling_slice(df_demand_profiles, t_week_and_day_pair).values) * 
            (0.001 if units == 'GWh' else 1))


def get_peak_demand(df_demand_profiles: pd.DataFrame=None,
                    t_week_and_day_pair: Optional[Tuple[Union[int, None], Union[int, None]]]=None) -> Number:
    """ Time-period flexible means of computing peak demand.

    Keyword Args:
        df_demand_profiles(pd.DataFrame):

def _get_df_profiling_slice(df_demand_profiles: pd.DataFrame=None,
                            t_week_and_day_pair: Optional[Tuple[Union[int, None], Union[int, None]]]=None) -> pd.DataFrame:
    """ Time-period flexible means of extracting a slice of the demand profile DataFrame.

    Keyword Args:
        df_demand_profiles(pd.DataFrame):
            - 
        t_week_and_day_pair(Optional[Tuple[Union[int, None], Union[int, None]]]):
            - A pair of one-indexed values, denoting Wk-of-the-Year, and Day-of-the-Wk, respectively. 
            - If the Tuple is set to None, the method defaults to annualised data. 
    Return: 
        pd.DataFrame: """
    import re
    # Construct the regular expression, based on `wk01d01` stylised column names
    if t_week_and_day_pair is None:  # Default ot Annual Maximum
        col_select_regexp = '^wk*'   # i.e. process all columns for the year
    else:
        _d_date_keys = dict()
        _d_date_keys.update({_k: (None if _x is None 
                                  else ("0%s" % _x if _x < 10 
                                        else "%s" % _x)) for _k, _x in zip(['wk_of_yr', 'day_of_wk'], 
                                                                           t_week_and_day_pair)})
        col_select_regexp = (r'^wk\d+' if _d_date_keys['wk_of_yr'] is None else "^wk%s" % _d_date_keys['wk_of_yr'])
        col_select_regexp += ("" if _d_date_keys['day_of_wk'] is None else "d%s$" % _d_date_keys['day_of_wk'])
        # logs.debug("\'col_select_regexp\':\t%s" % col_select_regexp)

    _l_matched_col_names = list()
    _l_matched_col_names.extend([_ for _ in df_demand_profiles.columns.values if re.match(col_select_regexp, _) is not None])
    logs.debug("\'_l_matched_col_names\':\t%s" % _l_matched_col_names)
    return df_demand_profiles.loc[:, df_demand_profiles.columns.isin(_l_matched_col_names)]
