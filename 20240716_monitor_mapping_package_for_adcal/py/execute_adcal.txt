def import_adcal_files(
        filepath_to_config: str=None) -> MonitorMappingOfAssetFamilyBase:
    with open(filepath_to_config) as _hndl_json: 
        d_sim_cfg = json.loads(_hndl_json.read())
    
    logs.info("%s", import_adcal_files.__name__)

    # paths to AdCal CSV files
    f_adcal_load_mdl = os.path.join(*([os.path.abspath("..\\")] + 
                                      d_sim_cfg['adcal']['input_data']['load_model']['subpath'] + 
                                      [d_sim_cfg['adcal']['input_data']['load_model']['demand_profiles']]))
    f_adcal_genshapes = os.path.join(*([os.path.abspath("..\\")] + 
                                       d_sim_cfg['adcal']['input_data']['gen_model']['subpath'] + 
                                       [d_sim_cfg['adcal']['input_data']['gen_model']['ambient_temperature_shapes']]))  # Genshapes_for_AdCal.csv

    return f_adcal_load_mdl, f_adcal_genshapes


def import_monitor_map_from_json(
        filepath_to_config: str=None) -> MonitorMappingOfAssetFamilyBase:
    with open(filepath_to_config) as _hndl_json: 
        d_sim_cfg = json.loads(_hndl_json.read())

    # path for initiating the export of the monitor map
    f_mapfile = os.path.join(*([os.path.abspath("..\\models\\circuit\\")] + 
                               d_sim_cfg['adcal']['input_data']['producers_map']['subpath'] + 
                               [d_sim_cfg['adcal']['input_data']['producers_map']['mapfile']]))
    logs.debug("Path for JSON import:\t %s\n", f_mapfile)

    with open(f_mapfile, 'r') as _hndl_map_json_ip:
        map_as_dict = json.loads(_hndl_map_json_ip.read())
    logs.debug("Check that imported content is of the correct format ...\t\'%s\'" % type(map_as_dict))

    # o_producers_map_as_base = MonitorMappingOfAssetFamilyBase.deserialize_from_json(json.loads(map_as_json_str))
    o_producers_map_as_base = MonitorMappingOfAssetFamilyBase.deserialize_from_json(map_as_dict)
    
    logs.debug("%s:\t%s" % (type(o_producers_map_as_base.subgroups['wind_ons']), 
                            o_producers_map_as_base.subgroups['wind_ons'].get_datatable()))  # get_collection_of_mapped_monitors()))
    logs.critical("%s:\t%s" % (type(o_producers_map_as_base), 
                               o_producers_map_as_base.get_datatable()))  # get_collection_of_mapped_monitors()))
    
    return o_producers_map_as_base
